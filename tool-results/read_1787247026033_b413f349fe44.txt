   251→  // ──────────────────────────────────────────────────────────────────────
   252→  // P11: Recipient Inbox (ARCH-038, ARCH-039, ARCH-040).
   253→  //
   254→  // When a bundle reaches DELIVERED at a node, CommOS auto-decrypts it
   255→  // using the node's encryption secret key and stores it in the inbox.
   256→  // The inbox is grouped by conversation_id for the unified-inbox UI.
   257→  // ──────────────────────────────────────────────────────────────────────
   258→
   259→  /**
   260→   * P11: handle a bundle that reached DELIVERED at a node. Auto-decrypts
   261→   * the bundle using the node's encryption secret key and adds it to the inbox.
   262→   */
   263→  handleDelivered(node_id: string, bundle: CommunicationBundle, from_node_id: string): void {
   264→    const entry = this.identities.get(node_id);
   265→    if (!entry) return;
   266→    // Only inbox bundles addressed to THIS node's identity.
   267→    if (bundle.recipient.kind !== 'IDENTITY' || bundle.recipient.ref.id !== entry.identity.id) return;
   268→    // Don't double-add (dedup by bundle_id).
   269→    const existing = this.inbox.get(node_id);
   270→    if (existing && existing.some((m) => m.bundle_id === bundle.bundle_id)) return;
   271→
   272→    // Decrypt the bundle.
   273→    const env = { encryption_metadata: bundle.encryption_metadata, payload: bundle.payload };
   274→    if (!isRecipientFor(env, entry.identity.public_keys.encryption_pubkey)) return;
   275→    let plaintext: string;
   276→    try {
   277→      const pt = openSealedPayload(env, entry.encryption_sk);
   278→      plaintext = utf8Decode(pt);
   279→    } catch {
   280→      return; // decryption failed — don't inbox
   281→    }
   282→
   283→    const msg: InboxMessage = {
   284→      bundle_id: bundle.bundle_id,
   285→      conversation_id: bundle.conversation_id,
   286→      sender: {
   287→        id: bundle.sender.id,
   288→        signing_pubkey_hash: bundle.sender.signing_pubkey_hash,
   289→        display_name: bundle.sender.display_name,
   290→      },
   291→      plaintext,
   292→      received_at: Date.now(),
   293→      read: false,
   294→      delivery_state: 'DELIVERED',
   295→      from_node_id,
   296→    };
   297→
   298→    if (!this.inbox.has(node_id)) this.inbox.set(node_id, []);
   299→    this.inbox.get(node_id)!.push(msg);
   300→  }
   301→
   302→  /** P11: get a node's inbox, grouped by conversation_id. */
   303→  getInbox(node_id: string): Array<{ conversation_id: string; messages: InboxMessage[]; unread_count: number }> {
   304→    const messages = this.inbox.get(node_id) ?? [];
   305→    // Group by conversation_id.
   306→    const byConv = new Map<string, InboxMessage[]>();
   307→    for (const msg of messages) {
   308→      if (!byConv.has(msg.conversation_id)) byConv.set(msg.conversation_id, []);
   309→      byConv.get(msg.conversation_id)!.push(msg);
   310→    }
   311→    // Sort conversations by most recent message.
   312→    const conversations = Array.from(byConv.entries()).map(([conv_id, msgs]) => ({
   313→      conversation_id: conv_id,
   314→      messages: msgs.sort((a, b) => a.received_at - b.received_at),
   315→      unread_count: msgs.filter((m) => !m.read).length,
   316→    }));
   317→    conversations.sort((a, b) => {
   318→      const aLast = a.messages[a.messages.length - 1]?.received_at ?? 0;
   319→      const bLast = b.messages[b.messages.length - 1]?.received_at ?? 0;
   320→      return bLast - aLast;
   321→    });
   322→    return conversations;
   323→  }

   325→  /**
   326→   * P11: mark all messages in a conversation as read. Also transitions the
   327→   * delivery state machine to READ for each bundle.
   328→   */
   329→  markConversationRead(node_id: string, conversation_id: string): { ok: boolean; marked: number } {
   330→    const messages = this.inbox.get(node_id);
   331→    if (!messages) return { ok: false, marked: 0 };
   332→    let marked = 0;
   333→    const rt = this.runtimes.get(node_id);
   334→    for (const msg of messages) {
   335→      if (msg.conversation_id === conversation_id && !msg.read) {
   336→        msg.read = true;
   337→        msg.delivery_state = 'READ';
   338→        marked++;
   339→        // Transition the delivery state machine to READ.
   340→        if (rt) {
   341→          try { rt.delivery.transition(msg.bundle_id, 'READ', { node: node_id }); } catch { /* already terminal */ }
   342→        }
   343→      }
   344→    }
   345→    return { ok: true, marked };
   346→  }

   352→  /**
   353→   * P12: compute delivery analytics from the delivery tracker + dispatched
   354→   * bundles. Per THREAT_MODEL §11 (Observability): does NOT expose private
   355→   * message contents — only aggregate statistics.
   356→   */
   357→  async getAnalytics(): Promise<{
   358→    total_dispatched: number;
   359→    total_delivered: number;
   360→    total_expired: number;
   361→    total_no_route: number;
   362→    total_relayed: number;
   363→    total_queued: number;
   364→    delivery_rate: number;
   365→    per_node: Array<{
   366→      node_id: string;
   367→      delivered: number;
   368→      relayed: number;
   369→      expired: number;
   370→      no_route: number;
   371→      queued: number;
   372→    }>;
   373→    route_stats: {
   374→      avg_reliability: number;
   375→      avg_latency_ms: number;
   376→      avg_cost: number;
   377→      hop_distribution: Record<number, number>;
   378→    };
   379→  }> {
   380→    const snapshots = await this.deliverySnapshots();
   381→    const dispatched = Array.from(this.dispatchedBundles.values());
   382→
   383→    let totalDelivered = 0;
   384→    let totalExpired = 0;
   385→    let totalNoRoute = 0;
   386→    let totalRelayed = 0;
   387→    let totalQueued = 0;
   388→
   389→    const perNodeMap = new Map<string, { delivered: number; relayed: number; expired: number; no_route: number; queued: number }>();
   390→
   391→    for (const snap of snapshots) {
   392→      const state = snap.current;
   393→      const nodeEntry = perNodeMap.get(snap.node_id) ?? { delivered: 0, relayed: 0, expired: 0, no_route: 0, queued: 0 };
   394→      if (state === 'DELIVERED' || state === 'READ') { totalDelivered++; nodeEntry.delivered++; }
   395→      else if (state === 'EXPIRED') { totalExpired++; nodeEntry.expired++; }
   396→      else if (state === 'NO_ROUTE') { totalNoRoute++; nodeEntry.no_route++; }
   397→      else if (state === 'RELAYED') { totalRelayed++; nodeEntry.relayed++; }
   398→      else if (state === 'QUEUED') { totalQueued++; nodeEntry.queued++; }
   399→      perNodeMap.set(snap.node_id, nodeEntry);
   400→    }
   401→
   402→    // Route stats from dispatched bundles (if they have a route plan).
   403→    let sumReliability = 0;
   404→    let sumLatency = 0;
   405→    let sumCost = 0;
   406→    let routeCount = 0;
   407→    const hopDist: Record<number, number> = {};
   408→    for (const d of dispatched) {
   409→      // We don't store the route plan with the dispatched bundle; compute from delivery snapshots.
   410→      // For now, count hops from the number of delivery records per bundle_id.
   411→      const bundleSnaps = snapshots.filter((s) => s.bundle_id === d.bundle.bundle_id);
   412→      const hops = bundleSnaps.length;
   413→      hopDist[hops] = (hopDist[hops] ?? 0) + 1;
   414→      routeCount++;
   415→    }
   416→
   417→    return {
   418→      total_dispatched: dispatched.length,
   419→      total_delivered: totalDelivered,
   420→      total_expired: totalExpired,
   421→      total_no_route: totalNoRoute,
   422→      total_relayed: totalRelayed,
   423→      total_queued: totalQueued,
   424→      delivery_rate: dispatched.length > 0 ? totalDelivered / dispatched.length : 0,
   425→      per_node: Array.from(perNodeMap.entries()).map(([node_id, stats]) => ({ node_id, ...stats })),
   426→      route_stats: {
   427→        avg_reliability: routeCount > 0 ? sumReliability / routeCount : 0,
   428→        avg_latency_ms: routeCount > 0 ? sumLatency / routeCount : 0,
   429→        avg_cost: routeCount > 0 ? sumCost / routeCount : 0,
   430→        hop_distribution: hopDist,
   431→      },
   432→    };
   433→  }

   455→  //
   456→  // Per master prompt §19: "Never trust self-reported contribution.
   457→  // Contribution accounting must eventually rely upon verifiable evidence.

   459→  //
   460→  // P13 measures participation from the delivery tracker's per-node state
   461→  // transitions — these are OBSERVABLE (the tracker records actual state
   462→  // changes per node per bundle). True VERIFIABLE evidence would come from

   464→  // aren't currently stored centrally. The measurement is a first step;

   466→  //
   467→  // NO tokens, NO credits, NO economy — just measurement + reputation.
   468→  // ──────────────────────────────────────────────────────────────────────

   470→  /**
   471→   * P13: compute community network statistics from the delivery tracker.
   472→   * Per-node: how many bundles each node relayed, delivered, gateway-handled.
   473→   * Reputation: delivery success rate (DELIVERED / (DELIVERED + EXPIRED + NO_ROUTE)).
   474→   * Resource accounting: from the capability cache (battery, bandwidth, storage).
   475→   */
   476→  async getCommunityStats(): Promise<{
   477→    per_node: Array<{
   478→      node_id: string;
   479→      display_name: string;
   480→      roles: string[];
   481→      relayed: number;
   482→      delivered: number;
   483→      gateway_handled: number;
   484→      expired: number;
   485→      no_route: number;
   486→      reputation: number;
   487→      resource: { battery_pct?: number; bandwidth_bps?: number; storage_bytes?: number };
   488→      verification: string;
   489→    }>;
   490→    total_relays: number;
   491→    total_deliveries: number;
   492→    total_gateway_handling: number;
   493→    total_expired: number;
   494→    network_reliability: number;
   495→    /** Distinction between observable and verifiable evidence. */
   496→    evidence_note: string;
   497→  }> {
   498→    const snapshots = await this.deliverySnapshots();
   499→    const nodeEntries = this.listNodes();
   500→
   501→    const perNodeMap = new Map<string, {
   502→      relayed: number; delivered: number; gateway_handled: number; expired: number; no_route: number;
   503→    }>();
   504→
   505→    for (const snap of snapshots) {
   506→      const entry = perNodeMap.get(snap.node_id) ?? { relayed: 0, delivered: 0, gateway_handled: 0, expired: 0, no_route: 0 };
   507→      const state = snap.current;
   508→      if (state === 'DELIVERED' || state === 'READ') entry.delivered++;
   509→      else if (state === 'RELAYED') entry.relayed++;
   510→      else if (state === 'GATEWAY_REACHED' || state === 'EXTERNAL_ACCEPTED') entry.gateway_handled++;
   511→      else if (state === 'EXPIRED') entry.expired++;
   512→      else if (state === 'NO_ROUTE') entry.no_route++;
   513→      perNodeMap.set(snap.node_id, entry);
   514→    }
   515→
   516→    let totalRelays = 0;
   517→    let totalDeliveries = 0;
   518→    let totalGatewayHandling = 0;
   519→    let totalExpired = 0;
   520→
   521→    const perNode = nodeEntries.map((n) => {
   522→      const stats = perNodeMap.get(n.node_id) ?? { relayed: 0, delivered: 0, gateway_handled: 0, expired: 0, no_route: 0 };
   523→      totalRelays += stats.relayed;
   524→      totalDeliveries += stats.delivered;
   525→      totalGatewayHandling += stats.gateway_handled;
   526→      totalExpired += stats.expired;
   527→      // Reputation: delivery success rate (DELIVERED / (DELIVERED + EXPIRED + NO_ROUTE)).
   528→      const totalAttempts = stats.delivered + stats.expired + stats.no_route;
   529→      const reputation = totalAttempts > 0 ? stats.delivered / totalAttempts : 1.0;
   530→      return {
   531→        node_id: n.node_id,
   532→        display_name: n.display_name,
   533→        roles: n.roles,
   534→        relayed: stats.relayed,
   535→        delivered: stats.delivered,
   536→        gateway_handled: stats.gateway_handled,
   537→        expired: stats.expired,
   538→        no_route: stats.no_route,
   539→        reputation,
   540→        resource: {
   541→          battery_pct: n.capabilities.resource.battery_pct,
   542→          bandwidth_bps: n.capabilities.resource.bandwidth_bps,
   543→          storage_bytes: n.capabilities.resource.storage_bytes,
   544→        },
   545→        verification: n.capabilities.verification,
   546→      };
   547→    });
   548→
   549→    const totalAttempts = totalDeliveries + totalExpired;
   550→    const networkReliability = totalAttempts > 0 ? totalDeliveries / totalAttempts : 1.0;
   551→
   552→    return {
   553→      per_node: perNode.sort((a, b) => (b.relayed + b.delivered + b.gateway_handled) - (a.relayed + a.delivered + a.gateway_handled)),
   554→      total_relays: totalRelays,
   555→      total_deliveries: totalDeliveries,
   556→      total_gateway_handling: totalGatewayHandling,
   557→      total_expired: totalExpired,
   558→      network_reliability: networkReliability,
   559→      evidence_note: 'Observable: per-node delivery state transitions. Verifiable: RELAY_FORWARD proofs (Ed25519-signed). NO tokens, NO credits — measurement only per ARCH-048.',
   560→    };
   561→  }

   748→    const nodes = this.listNodes().map((n) => ({

   754→    const seen = new Set<string>();

   757→        const peerSet: Set<string> = (t as any).peers ?? new Set();
   758→        for (const p of peerSet) {
   759→          const key = [nodeId, p].sort().join('|') + '|' + t.transport_type;
   760→          if (seen.has(key)) continue;
   761→          seen.add(key);
   762→          links.push({ from: nodeId, to: p, transport: t.transport_type });
   763→        }
   764→      }
   765→    }
   766→    const capabilities: Record<string, NodeCapabilities> = {};
   767→    for (const n of this.listNodes()) capabilities[n.node_id] = n.capabilities;
   768→    return { nodes, links, capabilities };

   771→  async dispatch(req: DispatchRequest): Promise<DispatchResponse> {
   772→    const senderRT = this.runtimes.get(req.from_node_id);
   773→    if (!senderRT) return { status: 'ERROR', error: `Unknown from_node_id: ${req.from_node_id}` };
   774→    if (!req.to_node_id && !req.to_channel) {
   775→      return { status: 'ERROR', error: 'Must specify either to_node_id or to_channel' };
   776→    }
   777→    if (req.to_node_id && req.to_channel) {
   778→      return { status: 'ERROR', error: 'Cannot specify both to_node_id and to_channel' };
   779→    }
   780→    const senderEntry = this.identities.get(req.from_node_id)!;
   781→
   782→    // P6+P10: For channel recipients, the bundle's payload is end-to-end encrypted
   783→    // to the recipient's REAL X25519 pubkey, looked up via the IdentityGraph.
   784→    // AUDIT-FIX: the synthesizeChannelIdentity() fallback has been REMOVED.
   785→    // If no verified IdentityGraph link exists, dispatch is REFUSED (ARCH-034).
   786→    let recipientEncryptionPubkey: Uint8Array;
   787→    let recipientRef: { id: string; signing_pubkey_hash: string; display_name?: string };
   788→    let recipientDescriptor: string;
   789→    let conversationId: string;
   790→    let destinationInput: { node_id?: string; channel?: string; channel_id?: string } | undefined;
   791→
   792→    if (req.to_node_id) {
   793→      const recipientEntry = this.identities.get(req.to_node_id);
   794→      if (!recipientEntry) return { status: 'ERROR', error: `Unknown to_node_id: ${req.to_node_id}` };
   795→      recipientEncryptionPubkey = recipientEntry.identity.public_keys.encryption_pubkey;
   796→      recipientRef = toRef(recipientEntry.identity);
   797→      recipientDescriptor = recipientEntry.identity.id;
   798→      conversationId = req.conversation_id ?? `conv:${req.from_node_id}:${req.to_node_id}`;
   799→      destinationInput = { node_id: req.to_node_id };
   800→    } else {
   801→      // P10+AUDIT-FIX: Channel recipient. Look up the recipient's real pubkey via
   802→      // the IdentityGraph. If no verified link exists, REFUSE to send.
   803→      // Per ARCH-034: "The caller MUST NOT encrypt to an unverified recipient."
   804→      // The previous code fell back to synthesizeChannelIdentity() which derived
   805→      // a deterministic keypair from a public string — NOT real E2E encryption.
   806→      // That fallback has been removed.
   807→      const resolved = this.resolveChannelRecipient(req.to_channel!.channel, req.to_channel!.channel_id);
   808→      if (!resolved) {
   809→        return {
   810→          status: 'ERROR',
   811→          error: `No verified identity link for ${req.to_channel!.channel}:${req.to_channel!.channel_id}. ` +
   812→                 `Per ARCH-034, the dispatcher refuses to encrypt to an unverified recipient. ` +
   813→                 `Link this channel identity via the Identity Graph first.`,
   814→        };
   815→      }
   816→      recipientEncryptionPubkey = resolved.encryption_pubkey;
   817→      recipientRef = resolved.identity_ref;
   818→      recipientDescriptor = `${req.to_channel!.channel}:${req.to_channel!.channel_id}`;
   819→      conversationId = req.conversation_id ?? `conv:${req.from_node_id}:${recipientDescriptor}`;
   820→      destinationInput = {
   821→        channel: req.to_channel!.channel,
   822→        channel_id: req.to_channel!.channel_id,
   823→      };
   824→    }
   825→
   826→    const intent = createIntent({
   827→      type: req.intent_type,
   828→      priority: req.priority,
   829→      ttl_ms: 60_000,
   830→    });
   831→
   832→    // Build the bundle first (so we have the canonical bundle_id), then seal.
   833→    const now = Date.now();
   834→    const expires_at = now + 60_000;
   835→
   836→    // To seal, we need a placeholder bundle_id; we'll re-seal after createBundle
   837→    // assigns the real UUID (canonical envelope binds bundle_id into additional_data).
   838→    const tempEnvelope = sealPayload({
   839→      bundle_id: 'pending',
   840→      intent_type: intent.type,
   841→      expires_at,
   842→      sender: toRef(senderEntry.identity),
   843→      recipient_encryption_pubkey: recipientEncryptionPubkey,
   844→      plaintext: utf8Encode(req.plaintext),
   845→    });
   846→
   847→    const recipient = req.to_node_id
   848→      ? { kind: 'IDENTITY' as const, ref: toRef(this.identities.get(req.to_node_id)!.identity) }
   849→      : { kind: 'CHANNEL' as const, channel: req.to_channel!.channel, channel_id: req.to_channel!.channel_id };
   850→
   851→    const bundle = createBundle({
   852→      sender: toRef(senderEntry.identity),
   853→      recipient,
   854→      conversation_id: conversationId,
   855→      intent,
   856→      encryption_metadata: tempEnvelope.encryption_metadata,
   857→      payload: tempEnvelope.payload,
   858→      created_at: now,
   859→      expires_at,
   860→      routing_policy: {
   861→        policy_id: defaultPolicy.policy_id,
   862→        inline: {
   863→          replication_factor: req.replicate ? 3 : 1,
   864→          max_hops: defaultPolicy.max_hops,
   865→          require_e2e: defaultPolicy.require_e2e,
   866→        },
   867→      },
   868→    });
   869→
   870→    // Re-seal with the real bundle_id so the AEAD additional_data binding matches.
   871→    const realEnv = sealPayload({
   872→      bundle_id: bundle.bundle_id,
   873→      intent_type: intent.type,
   874→      expires_at: bundle.expires_at,
   875→      sender: toRef(senderEntry.identity),
   876→      recipient_encryption_pubkey: recipientEncryptionPubkey,
   877→      plaintext: utf8Encode(req.plaintext),
   878→    });
   879→    const signedBundle: CommunicationBundle = {
   880→      ...bundle,
   881→      encryption_metadata: realEnv.encryption_metadata,
   882→      payload: realEnv.payload,
   883→    };
   884→
   885→    const senderProof = signProof(
   886→      'SENDER_SIGNATURE',
   887→      {
   888→        bundle_id: signedBundle.bundle_id,
   889→        sender_id: signedBundle.sender.id,
   890→        sender_signing_pubkey_hash: signedBundle.sender.signing_pubkey_hash,
   891→        recipient_kind: signedBundle.recipient.kind,
   892→        recipient_descriptor: recipientDescriptor,
   893→        conversation_id: signedBundle.conversation_id,
   894→        intent_type: signedBundle.intent.type,
   895→        priority: signedBundle.intent.priority,
   896→        created_at: signedBundle.created_at,
   897→        expires_at: signedBundle.expires_at,
   898→        algorithm: signedBundle.encryption_metadata.algorithm,
   899→        recipient_pubkey_hash: signedBundle.encryption_metadata.recipient_pubkey_hash,
   900→        nonce: signedBundle.encryption_metadata.nonce,
   901→        additional_data: signedBundle.encryption_metadata.additional_data,
   902→        payload_bytes_len: signedBundle.payload.bytes_len,
   903→        routing_policy_id: signedBundle.routing_policy.policy_id,
   904→        replication_factor: signedBundle.routing_policy.inline.replication_factor,
   905→        max_hops: signedBundle.routing_policy.inline.max_hops,
   906→        require_e2e: signedBundle.routing_policy.inline.require_e2e,
   907→      },
   908→      toRef(senderEntry.identity),
   909→      senderEntry.signing_sk,
   910→    );
   911→
   912→    const bundleWithProof: CommunicationBundle = {
   913→      ...signedBundle,
   914→      proofs: [senderProof],
   915→    };
   916→
   917→    this.dispatchedBundles.set(bundleWithProof.bundle_id, {
   918→      bundle: bundleWithProof,
   919→      from_node_id: req.from_node_id,
   920→      plaintext: req.plaintext,
   921→    });
   922→
   923→    const result = await senderRT.dispatch({
   924→      bundle: bundleWithProof,
   925→      destination: destinationInput,
   926→      replicate: req.replicate,
   927→    });
   928→
   929→    return {
   930→      status: result.status,
   931→      bundle_id: bundleWithProof.bundle_id,
   932→      replicas_sent: result.replicas_sent,
   933→      route_plan: result.plan
   934→        ? {
   935→            hops: result.plan.hops.map((h) => ({
   936→              kind: h.kind,
   937→              to_node_id: h.to_node_id,
   938→              transport: h.transport,
   939→              gateway: h.gateway,
   940→              est_reliability: h.est_reliability,
   941→              est_latency_ms: h.est_latency_ms,
   942→              est_cost: h.est_cost,
   943→            })),
   944→            rationale: result.plan.rationale,
   945→            est_reliability: result.plan.est_reliability,
   946→            est_latency_ms: result.plan.est_latency_ms,
   947→            est_cost: result.plan.est_cost,
   948→          }
   949→        : undefined,
   950→      error: result.error,
   951→    };
   952→  }

   969→  /**
   970→   * Snapshot delivery state across all nodes. P3: now per (bundle, node).
   971→   * In persistent mode, the snapshot comes from the DB; in-memory mode reads
   972→   * from each runtime's tracker.
   973→   */
   974→  async deliverySnapshots(): Promise<DeliverySnapshot[]> {
   975→    // The in-memory per-node delivery tracker is the LIVE source of truth
   976→    // for the delivery state machine (ARCH-012). The Prisma-backed tracker
   977→    // is for forensic/restart recovery — keep them in sync via the sweeper.
   978→    const out: DeliverySnapshot[] = [];
   979→    for (const rt of this.runtimes.values()) {
   980→      for (const r of rt.delivery.snapshot()) {
   981→        out.push({

  1000→  tryDecrypt(bundle_id: string, at_node_id: string): { ok: boolean; plaintext?: string; reason?: string } {
  1001→    const entry = this.dispatchedBundles.get(bundle_id);
  1002→    if (!entry) return { ok: false, reason: 'unknown bundle' };
  1003→    const nodeEntry = this.identities.get(at_node_id);
  1004→    if (!nodeEntry) return { ok: false, reason: 'unknown node' };
  1005→
  1006→    const env = { encryption_metadata: entry.bundle.encryption_metadata, payload: entry.bundle.payload };
  1007→    if (!isRecipientFor(env, nodeEntry.identity.public_keys.encryption_pubkey)) {
  1008→      return { ok: false, reason: 'this node is not the recipient (recipient_pubkey_hash mismatch)' };
  1009→    }
  1010→    try {
  1011→      const pt = openSealedPayload(env, nodeEntry.encryption_sk);
  1012→      return { ok: true, plaintext: utf8Decode(pt) };
  1013→    } catch (e) {
  1014→      return { ok: false, reason: String(e) };
  1015→    }
  1016→  }

  1039→  /**
  1040→   * Return the proof chain for a bundle: SENDER_SIGNATURE + any RELAY_FORWARD proofs
  1041→   * appended by relays. Verifies each proof against the corresponding signer's public key.
  1042→   */
  1043→  async relayForwardProofs(bundle_id: string): Promise<RelayForwardProofView | undefined> {
  1044→    const entry = this.dispatchedBundles.get(bundle_id);
  1045→    if (!entry) return undefined;
  1046→    // AUDIT-FIX: The original code stored only the original dispatched bundle
  1047→    // (with SENDER_SIGNATURE only). RELAY_FORWARD proofs are added by relays
  1048→    // during forwarding — they're in the RECEIVED bundle, not the dispatched one.
  1049→    // The dispatched bundle's proof chain only has the SENDER_SIGNATURE.
  1050→    // For honest verification, we verify the SENDER_SIGNATURE with full canonical
  1051→    // fields. RELAY_FORWARD proofs from the forwarded bundle aren't available
  1052→    // here — they'd need to be captured from the recipient's inbox.
  1053→    const proofs = entry.bundle.proofs ?? [];
  1054→
  1055→    const verifiedProofs = proofs.map((p) => {
  1056→      let verified = false;
  1057→      for (const { identity } of this.identities.values()) {
  1058→        if (identity.signing_pubkey_hash === p.signer.signing_pubkey_hash) {
  1059→          try {
  1060→            if (p.kind === 'SENDER_SIGNATURE') {
  1061→              // AUDIT-FIX: use full canonical fields, not approximate.
  1062→              verified = verifyProof(p, {
  1063→                bundle_id: entry.bundle.bundle_id,
  1064→                sender_id: entry.bundle.sender.id,
  1065→                sender_signing_pubkey_hash: entry.bundle.sender.signing_pubkey_hash,
  1066→                recipient_kind: entry.bundle.recipient.kind,
  1067→                recipient_descriptor: entry.bundle.recipient.kind === 'IDENTITY'
  1068→                  ? entry.bundle.recipient.ref.id : 'other',
  1069→                conversation_id: entry.bundle.conversation_id,
  1070→                intent_type: entry.bundle.intent.type,
  1071→                priority: entry.bundle.intent.priority,
  1072→                created_at: entry.bundle.created_at,
  1073→                expires_at: entry.bundle.expires_at,
  1074→                algorithm: entry.bundle.encryption_metadata.algorithm,
  1075→                recipient_pubkey_hash: entry.bundle.encryption_metadata.recipient_pubkey_hash,
  1076→                nonce: entry.bundle.encryption_metadata.nonce,
  1077→                additional_data: entry.bundle.encryption_metadata.additional_data,
  1078→                payload_bytes_len: entry.bundle.payload.bytes_len,
  1079→                routing_policy_id: entry.bundle.routing_policy.policy_id,
  1080→                replication_factor: entry.bundle.routing_policy.inline.replication_factor,
  1081→                max_hops: entry.bundle.routing_policy.inline.max_hops,
  1082→                require_e2e: entry.bundle.routing_policy.inline.require_e2e,
  1083→              }, identity.public_keys.signing_pubkey);
  1084→            } else {
  1085→              // RELAY_FORWARD: can't verify without the forwarded bundle's proof chain.
  1086→              // Mark as unverified with a note.
  1087→              verified = false;
  1088→            }
  1089→          } catch {
  1090→            verified = false;
  1091→          }
  1092→          break;
  1093→        }
  1094→      }
  1095→      return {
  1096→        kind: p.kind,
  1097→        signer_id: p.signer.id,
  1098→        ts: p.ts,
  1099→        verified,
  1100→      };
  1101→    });
  1102→
  1103→    return {
  1104→      bundle_id,
  1105→      proofs: verifiedProofs,
  1106→    };
  1107→  }

  1109→  async sweepOnce() {
  1110→    const result = await this.sweeper.sweepOnce();
  1111→    // Mirror the EXPIRED transitions to each node's in-memory tracker so the
  1112→    // UI sees them (ARCH-022: dual-located delivery tracker).
  1113→    for (const rt of this.runtimes.values()) {
  1114→      try {
  1115→        // For each expired bundle in this node's store, mark EXPIRED in the
  1116→        // in-memory tracker if the bundle has a tracker entry and is in QUEUED state.
  1117→        for (const q of await rt.queuedBundles()) {
  1118→          const existing = rt.delivery.get(q.bundle.bundle_id);
  1119→          if (existing && existing.current === 'QUEUED') {
  1120→            try {
  1121→              rt.delivery.transition(q.bundle.bundle_id, 'EXPIRED', {
  1122→                node: rt.node_id,
  1123→                note: 'TTL sweeper',
  1124→              });
  1125→            } catch {
  1126→              // State machine rejected — likely already terminal. Skip.
  1127→            }
  1128→          }
  1129→        }
  1130→      } catch {
  1131→        // Runtime store may be in-memory only (no queuedBundles persistence). Skip.
  1132→      }
  1133→    }
  1134→    return result;
  1135→  }
